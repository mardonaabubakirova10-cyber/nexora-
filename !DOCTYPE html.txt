          const rect = card.getBoundingClientRect();

          const x = e.clientX - rect.left;
          const y = e.clientY - rect.top;

          const rotateY = ((x / rect.width) - .5) * 5;
          const rotateX = ((y / rect.height) - .5) * -5;

          card.style.transform =

        });

      });


      /* ================= SEARCH ================= */

      const heroSearch = document.getElementById('heroSearch');
      const heroSearchBtn = document.getElementById('heroSearchBtn');

      const performSearch = () => {
        const value = heroSearch.value.trim().toLowerCase();

        document.getElementById('tutors').scrollIntoView({
          behavior: 'smooth'
        });

        setTimeout(() => {
          document.getElementById('searchInput')?.focus();

          if (value) {
            filterTutorsByText(value);
          }
        }, 500);
      };

      heroSearchBtn.addEventListener('click', performSearch);

      heroSearch.addEventListener('keydown', e => {
        if (e.key === 'Enter') performSearch();
      });


      /* ================= TUTOR FILTERS ================= */

      const cards = [...document.querySelectorAll('.tutor-item')];
      const subjectFilter = document.getElementById('subjectFilter');
      const priceFilter = document.getElementById('priceFilter');
      const sortFilter = document.getElementById('sortFilter');
      const emptyState = document.getElementById('emptyState');

      let activeLevel = 'all';
      let onlyOnline = false;
      let onlyFavorites = false;
      let textSearch = '';

      function getFavorites() {
        try {
          return JSON.parse(localStorage.getItem('nexoraFavorites')) || [];
        } catch {
          return [];
        }
      }

      function saveFavorites(items) {
        localStorage.setItem('nexoraFavorites', JSON.stringify(items));
      }

      function filterTutors() {

        let visible = [];

        cards.forEach(card => {

          const name = card.dataset.name.toLowerCase();
          const subject = card.dataset.subject;
          const price = Number(card.dataset.price);
          const rating = Number(card.dataset.rating);
          const level = card.dataset.level;
          const online = card.dataset.online === 'true';

          let matches = true;

          if (textSearch) {
            const fullText = (
              card.innerText +
              ' ' +
              card.dataset.name +
              ' ' +
              card.dataset.subject
            ).toLowerCase();

            if (!fullText.includes(textSearch)) {
              matches = false;
            }
          }

          if (subjectFilter.value !== 'all' &&
              subject !== subjectFilter.value) {
            matches = false;
          }

          if (priceFilter.value === 'low' && price > 1500) {
            matches = false;
          }

          if (priceFilter.value === 'mid' &&
              (price < 1500 || price > 2500)) {
            matches = false;
          }

          if (priceFilter.value === 'high' && price < 2500) {
            matches = false;
          }

          if (activeLevel !== 'all' && level !== activeLevel) {
            matches = false;
          }

          if (onlyOnline && !online) {
            matches = false;
          }

          if (onlyFavorites) {
            const favorites = getFavorites();

            if (!favorites.includes(card.dataset.name)) {
              matches = false;
            }
          }

          card.style.display = matches ? '' : 'none';

          if (matches) {
            visible.push({
              card,
              rating,
              price
            });
          }
        });

        const sort = sortFilter.value;

        visible.sort((a, b) => {

          if (sort === 'rating') {
            return b.rating - a.rating;
          }

          if (sort === 'priceAsc') {
            return a.price - b.price;
          }

          if (sort === 'priceDesc') {
            return b.price - a.price;
          }

          return 0;
        });

        visible.forEach(item => {
          document.getElementById('tutorsGrid').appendChild(item.card);
        });

        emptyState.classList.toggle('show', visible.length === 0);

        updateFavoriteButtons();
      }

      function filterTutorsByText(value) {
        textSearch = value;
        filterTutors();
      }

      subjectFilter.addEventListener('change', filterTutors);
      priceFilter.addEventListener('change', filterTutors);
      sortFilter.addEventListener('change', filterTutors);


      /* ================= FILTER CHIPS ================= */

      document.querySelectorAll('.filter-chip').forEach(chip => {

        chip.addEventListener('click', () => {

          document.querySelectorAll('.filter-chip')
            .forEach(item => item.classList.remove('active'));

          chip.classList.add('active');

          if (chip.dataset.level) {
            activeLevel = chip.dataset.level;
            onlyOnline = false;
            onlyFavorites = false;
          }

          if (chip.dataset.online) {
            onlyOnline = true;
            onlyFavorites = false;
            activeLevel = 'all';
          }

          if (chip.dataset.favorites) {
            onlyFavorites = true;
            onlyOnline = false;
            activeLevel = 'all';
          }

          filterTutors();
        });

      });


      /* ================= FILTER MOBILE ================= */

      const filterToggle = document.getElementById('filterToggle');
      const filtersPanel = document.getElementById('filtersPanel');

      filterToggle.addEventListener('click', () => {
        filtersPanel.classList.toggle('open');
      });


      /* ================= FAVORITES ================= */

      function updateFavoriteButtons() {

        const favorites = getFavorites();

        document.querySelectorAll('.favorite-btn').forEach(button => {

          const card = button.closest('.tutor-card');
          const name = card.dataset.name;
          const active = favorites.includes(name);

          button.classList.toggle('active', active);
          button.textContent = active ? '♥' : '♡';
        });
      }

      document.querySelectorAll('.favorite-btn').forEach(button => {

        button.addEventListener('click', e => {

          e.stopPropagation();

          const card = button.closest('.tutor-card');
          const name = card.dataset.name;

          let favorites = getFavorites();

          if (favorites.includes(name)) {
            favorites = favorites.filter(item => item !== name);
            showToast('Удалено из избранного');
          } else {
            favorites.push(name);
            showToast('Добавлено в избранное ♥');
          }

          saveFavorites(favorites);
          updateFavoriteButtons();

          if (onlyFavorites) {
            filterTutors();
          }
        });

      });

      updateFavoriteButtons();


      /* ================= TUTOR MODAL ================= */

      const modalOverlay = document.getElementById('modalOverlay');
      const modalClose = document.getElementById('modalClose');

      const modalTutorName = document.getElementById('modalTutorName');
      const modalTutorRole = document.getElementById('modalTutorRole');
      const modalAvatar = document.getElementById('modalAvatar');

      function openModal(tutor = '') {

        modalTutorName.textContent =
          tutor || 'Подбор преподавателя';

        modalTutorRole.textContent =
          tutor
            ? 'Запрос на пробное занятие'
            : 'Расскажите о своей цели';

        modalAvatar.textContent =
          tutor
            ? tutor.split(' ').map(word => word[0]).join('').slice(0,2)
            : 'NX';

        document.getElementById('bookingForm').style.display = '';
        document.getElementById('successMessage').classList.remove('show');

        modalOverlay.classList.add('open');
        document.body.classList.add('modal-open');

        setTimeout(() => {
          document.getElementById('name').focus();
        }, 250);
      }

      window.openModal = openModal;

      document.querySelectorAll('.open-tutor').forEach(button => {

        button.addEventListener('click', () => {

          const tutor = button.dataset.tutor;
          const role = button.dataset.role;
          const initials = button.dataset.initials;

          modalTutorName.textContent = tutor;
          modalTutorRole.textContent =
            role + ' · от ' + button.dataset.price;

          modalAvatar.textContent = initials;

          openModal(tutor);

          modalTutorRole.textContent =
            role + ' · от ' + button.dataset.price;
        });

      });

      function closeModal() {
        modalOverlay.classList.remove('open');
        document.body.classList.remove('modal-open');
      }

      modalClose.addEventListener('click', closeModal);

      modalOverlay.addEventListener('click', e => {
        if (e.target === modalOverlay) {
          closeModal();
        }
      });

      document.addEventListener('keydown', e => {
        if (e.key === 'Escape') {
          closeModal();
        }
      });


      /* ================= FORM ================= */

      const bookingForm = document.getElementById('bookingForm');
      const successMessage = document.getElementById('successMessage');

      bookingForm.addEventListener('submit', e => {

        e.preventDefault();

        bookingForm.style.display = 'none';
        successMessage.classList.add('show');

        showToast('Заявка успешно создана');

        setTimeout(() => {
          bookingForm.reset();
        }, 100);
      });


      /* ================= FAQ ================= */

      document.querySelectorAll('.faq-question').forEach(question => {

        question.addEventListener('click', () => {

          const item = question.closest('.faq-item');
          const wasActive = item.classList.contains('active');

          document.querySelectorAll('.faq-item')
            .forEach(faq => faq.classList.remove('active'));

          if (!wasActive) {
            item.classList.add('active');
          }
        });

      });


      /* ================= TOAST ================= */

      const toast = document.getElementById('toast');
      let toastTimer;

      function showToast(message) {

        toast.textContent = message;
        toast.classList.add('show');

        clearTimeout(toastTimer);

        toastTimer = setTimeout(() => {
          toast.classList.remove('show');
        }, 2200);
      }

      window.showToast = showToast;


      /* ================= LIVE HERO SEARCH ================= */

      const heroInput = document.getElementById('heroSearch');

      heroInput.addEventListener('input', () => {

        const value = heroInput.value.trim().toLowerCase();

        if (value.length >= 2) {
          textSearch = value;
        } else {
          textSearch = '';
        }
      });


      /* ================= SCROLL LINKS ================= */

      document.querySelectorAll('a[href^="#"]').forEach(link => {

        link.addEventListener('click', e => {

          const targetId = link.getAttribute('href');

          if (!targetId || targetId === '#') return;

          const target = document.querySelector(targetId);

          if (target) {
            e.preventDefault();

            target.scrollIntoView({
              behavior: 'smooth',
              block: 'start'
            });
          }
        });

      });

    });
  </script>
